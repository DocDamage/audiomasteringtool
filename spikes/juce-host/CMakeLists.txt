cmake_minimum_required(VERSION 3.24)
project(amt_juce8_host_spike VERSION 0.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_SOURCE_DIR "" CACHE PATH "Path to JUCE 8.0.15 source tree")
if(NOT JUCE_SOURCE_DIR)
  message(FATAL_ERROR "Set JUCE_SOURCE_DIR to JUCE 8.0.15")
endif()
add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
juce_add_gui_app(amt_juce8_host_spike PRODUCT_NAME "AMT JUCE 8 Host Spike")
target_sources(amt_juce8_host_spike PRIVATE Main.cpp)
target_compile_definitions(amt_juce8_host_spike PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(amt_juce8_host_spike PRIVATE juce::juce_gui_extra juce::juce_audio_utils)
